        var problems = LenoStartupConfigurationValidator.GetInvalidConfigurationKeys(_configuration, _serviceName);

                $"请通过 Consul KV（leno/config/**，生产主通道）或环境变量兜底注入：{string.Join("；", problems)}");
        }
